% ===================================================
% Reglas de inferencia lógica
% ===================================================

% El componente es ingrediente directo del objeto si aparece en la lista de receta del objeto
ingrediente_directo(Ingrediente, Objeto) :-
    receta(Objeto, Lista),
    member(Ingrediente, Lista).

% El componente es ingrediente indirecto recursivo del objeto
requiere_material(Ingrediente, Objeto) :-
    ingrediente_directo(Ingrediente, Objeto).
requiere_material(Ingrediente, Objeto) :-
    ingrediente_directo(Intermedio, Objeto),
    requiere_material(Ingrediente, Intermedio).

% ===================================================
% Comienzo del chatbot e interfaz de usuario
% ===================================================

iniciar_chatbot :-
    writeln("==================================================="),
    writeln("  Bienvenido al Chatbot de Crafteo de Minecraft    "),
    writeln("Puedes preguntarme cosas como:"),
    writeln(" - como hago el pico_hierro"),
    writeln(" - para que sirve la espada_diamante"),
    writeln(" - que necesito para hacha_madera"),
    writeln(" - para que sirve la pala_piedra"),
    writeln(" - salir"),
    writeln("---------------------------------------------------"),
    bucle_chat.

bucle_chat :-
    nl, write("Tú: "),
    read_line_to_string(user_input, Entrada),
    string_lower(Entrada, EntradaMinuscula),
    split_string(EntradaMinuscula, " ", "", Palabras),
    procesar_pregunta(Palabras).

% Salida del programa
procesar_pregunta(["salir"]) :-
    writeln("Chatbot: ¡Hasta luego! Buena suerte en tu mundo de Minecraft.").

% Preguntas sobre cómo craftear: "como hago" o "como crafteo"
procesar_pregunta(Palabras) :-
    (member("como", Palabras); member("crafteo", Palabras)),
    obtener_objeto(Palabras, Objeto),
    receta(Objeto, Ingredientes),
    !,
    format("Chatbot: Para hacer ~w necesitas: ~w.~n", [Objeto, Ingredientes]),
    bucle_chat.

% Preguntas sobre utilidad: "para que sirve" o "que hace"
procesar_pregunta(Palabras) :-
    (member("para", Palabras); member("sirve", Palabras)),
    obtener_objeto(Palabras, Objeto),
    uso(Objeto, Accion),
    !,
    format("Chatbot: El objeto ~w sirve para ~w.~n", [Objeto, Accion]),
    bucle_chat.

% Preguntas sobre materiales base: "que necesito para"
procesar_pregunta(Palabras) :-
    member("necesito", Palabras),
    obtener_objeto(Palabras, Objeto),
    !,
    findall(Mat, requiere_material(Mat, Objeto), Materiales),
    sort(Materiales, MatUnicos),
    format("Chatbot: Para construir ~w requerirás en la cadena de crafteo: ~w.~n", [Objeto, MatUnicos]),
    bucle_chat.

% Caso por defecto si no entiende la consulta
procesar_pregunta(_) :-
    writeln("Chatbot: No entendí tu pregunta. Intenta consultar sobre recetas (ej: \"como hago espada_hierro\") o uso (ej: \"para que sirve pala_madera\")."),
    bucle_chat.

% Auxiliar para extraer el nombre del objeto de la lista de palabras
obtener_objeto(Palabras, Objeto) :-
    member(PalabraStr, Palabras),
    atom_string(Objeto, PalabraStr),
    (receta(Objeto, _); uso(Objeto, _)).

:- initialization(iniciar_chatbot).
